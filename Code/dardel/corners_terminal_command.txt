pairs=("1 5" "5 1" "2 4" "4 2" "3 3" "1 4" "4 1" "2 3" "3 2" "1 3" "3 1" "2 2" "1 2" "2 1" "1 1")

for pair in "${pairs[@]}"; do
  M=$(echo "$pair" | cut -d' ' -f1)
  N=$(echo "$pair" | cut -d' ' -f2)
  for seed in {1..20}; do
    python3 CORNERS_B.py "$M" "$N" "$seed" \
      > "./corners/corners_${M}_${N}_${seed}.txt"
  done
done
